const BK = { greeks:new Map(), at:0, loading:false };
async function loadBookGreeks(){
  if (BK.loading) return; const pb = getPb(); const open = pb.positions.filter(p=>!p.closed); const syms = [...new Set(open.map(p=>p.sym))].filter(s => s !== S.sym);
  BK.loading = true; let changed = false;
  for (const sym of syms){ const cached = BK.greeks.get(sym); if (cached && Date.now() - cached.at < 10*60e3) continue; const chain = await fetchChainAny(sym); if (!chain && !S.demo) continue; const ch = chain || synthChain(sym); const exps = [...new Set(ch.rows.map(r=>r[0]))].sort().slice(0,12); const calc = computeFrom({...ch, symbol:sym}, exps, 15, S.set.model); BK.greeks.set(sym, {at:Date.now(), spot:calc.spot, contracts:calc.allContracts, demo:!chain}); changed = true; }
  BK.loading = false; if (changed && S.tab === "book") renderTabs();
}
function bookRows(){
  const pb = getPb(); const open = pb.positions.filter(p=>!p.closed); const rows = [];
  for (const p of open){ const src = p.sym === S.sym ? {spot:S.calc?.spot, contracts:S.calc?.allContracts||[], demo:S.demo} : BK.greeks.get(p.sym); if (!src || !src.spot){ rows.push({p, missing:true}); continue; }
    if (p.shares){ rows.push({p, spot:src.spot, mark:src.spot, delta$:p.qty*src.spot, theta$:0, vega$:0, stress:(pct) => p.qty*src.spot*pct, pnl:(src.spot-p.entry)*p.qty, demo:src.demo}); continue; }
    const x = src.contracts.find(y => y.exp===p.exp && y.K===p.K && y.type===p.cp); if (!x){ rows.push({p, missing:true}); continue; }
    const T0 = Math.max(0.5, dte(p.exp))/365, iv = x.iv||0.2, base = bsPrice(src.spot, p.K, T0, iv, p.cp), ratio = base ? x.mid/base : 1;
    const theta = (bsPrice(src.spot, p.K, Math.max(0,T0-1/365), iv, p.cp) - base)*ratio; const vega = (bsPrice(src.spot, p.K, T0, iv+0.01, p.cp) - base)*ratio;
    rows.push({p, spot:src.spot, mark:x.mid, delta$:x.delta*p.qty*100*src.spot, theta$:theta*p.qty*100, vega$:vega*p.qty*100, stress:(pct)=> (bsPrice(src.spot*(1+pct), p.K, T0, iv, p.cp)*ratio - x.mid)*p.qty*100, pnl:(x.mid-p.entry)*p.qty*100, demo:src.demo}); }
  return rows;
}

function coachReport(){
  const j = getJournal(); const closed = j.filter(e=>e.closed && e.pnl != null); const out = { lines:[], stats:{} };
  if (closed.length < 3){ out.lines.push("Log at least three closed trades and the coach has something to work with."); return out; }
  const wins = closed.filter(e=>e.pnl>0), losses = closed.filter(e=>e.pnl<=0);
  const avgW = wins.length ? wins.reduce((a,e)=>a+e.pnl,0)/wins.length : 0, avgL = losses.length ? losses.reduce((a,e)=>a+e.pnl,0)/losses.length : 0;
  const wr = wins.length/closed.length, expectancy = wr*avgW + (1-wr)*avgL; const pf = losses.length && avgL ? (wins.reduce((a,e)=>a+e.pnl,0))/Math.abs(losses.reduce((a,e)=>a+e.pnl,0)) : Infinity;
  const hold = e => (e.closed - e.opened)/86400000; const avgHoldW = wins.length ? wins.reduce((a,e)=>a+hold(e),0)/wins.length : 0, avgHoldL = losses.length ? losses.reduce((a,e)=>a+hold(e),0)/losses.length : 0;
  const bigLoss = losses.filter(e => e.pnlPct <= -25).length; const worst = closed.reduce((a,e)=>e.pnl<a.pnl?e:a, closed[0]); const best = closed.reduce((a,e)=>e.pnl>a.pnl?e:a, closed[0]);
  let streak = 0, maxLossStreak = 0; for (const e of closed.slice().sort((a,b)=>a.closed-b.closed)){ if (e.pnl <= 0){ streak++; maxLossStreak = Math.max(maxLossStreak, streak); } else streak = 0; }
  const bySetup = new Map(); for (const e of closed){ const k = (e.why||e.thesis||"unlabeled").split(/[:—-]/)[0].trim().slice(0,28).toLowerCase() || "unlabeled"; const v = bySetup.get(k) || {n:0,pnl:0,w:0}; v.n++; v.pnl += e.pnl; if (e.pnl>0) v.w++; bySetup.set(k,v); }
  const setups = [...bySetup.entries()].sort((a,b)=>b[1].pnl-a[1].pnl);
  const withSnap = closed.filter(e => e.snap && e.snap.pw && e.snap.cw); const aligned = withSnap.filter(e => { const s = e.snap; const isCall = /C\b|call/i.test(e.contract) && !/P\b|put/i.test(e.contract); return isCall ? (s.spot > (s.flip||0)) : (s.spot < (s.flip||Infinity)); }); const alignedWr = aligned.length ? aligned.filter(e=>e.pnl>0).length/aligned.length : null;
  const weekendHolds = closed.filter(e => { const o = new Date(e.opened), c = new Date(e.closed); return c - o > 2*86400000 && [5,6,0].some(d => { for (let t = o.getTime(); t <= c.getTime(); t += 86400000) if (new Date(t).getDay() === 6) return true; return false; }); }).length;
  out.stats = { n:closed.length, wr, avgW, avgL, expectancy, pf, avgHoldW, avgHoldL, maxLossStreak, bigLoss, weekendHolds, setups, alignedWr, alignedN:aligned.length };
  const L = out.lines;
  L.push(`${closed.length} closed trades, ${fmt(wr*100,0)}% win rate, average win ${money(avgW)} vs average loss ${money(avgL)}. Expectancy ${money(expectancy)} per trade; profit factor ${isFinite(pf)?fmt(pf,2):"∞"}.`);
  if (wr >= 0.5 && Math.abs(avgL) > avgW) L.push(`You win more often than you lose but your losers are bigger than your winners (${money(avgL)} vs ${money(avgW)}). That's the signature of cutting winners early and letting losers run. Your own rule — never let green go red — is doing the first half; the fix is the stop on the red side, honored at the invalidation, not at the premium.`);
  if (wr < 0.5 && avgW > Math.abs(avgL)*1.8) L.push(`Low hit rate, big winners: a runner's profile. It works as long as the losers stay small. Watch the loss streak — ${maxLossStreak} in a row is your record — and size so that streak can't dent the account.`);
  if (avgHoldL > avgHoldW*1.5 && losses.length >= 2) L.push(`You hold losers ${fmt(avgHoldL,1)} days on average vs ${fmt(avgHoldW,1)} for winners. Losers that need time are losers; the thesis-broken exit should come faster than the target.`);
  if (bigLoss) L.push(`${bigLoss} trade${bigLoss>1?"s":""} closed down 25% or more. With a 20% premium stop that shouldn't happen — either the stop isn't being placed or gaps are hitting it. If it's gaps, the fix is size, not a tighter stop.`);
  if (weekendHolds) L.push(`${weekendHolds} position${weekendHolds>1?"s":""} held over a weekend. For 30+ DTE that's fine; for anything short-dated it's your own rule being broken.`);
  if (setups.length >= 2){ const [bestS, worstS] = [setups[0], setups[setups.length-1]]; L.push(`By setup: "${bestS[0]}" is your best (${money(bestS[1].pnl)} over ${bestS[1].n}, ${bestS[1].w}/${bestS[1].n} wins)${worstS[1].pnl < 0 ? `; "${worstS[0]}" is costing you (${money(worstS[1].pnl)} over ${worstS[1].n})` : ""}. Do more of the first and demand a better reason for the second.`); }
  if (alignedWr != null && aligned.length >= 3){ const fought = withSnap.filter(e=>!aligned.includes(e)); const fWr = fought.length >= 2 ? fought.filter(e=>e.pnl>0).length/fought.length : null; L.push(`Trades taken with positioning on your side (right side of the flip at entry) won ${fmt(alignedWr*100,0)}% (${aligned.length} trades)${fWr != null ? ` vs ${fmt(fWr*100,0)}% when you fought it` : ""}. ${fWr == null || alignedWr >= fWr ? "That's the edge the Heatseeker is for — keep requiring it." : "The aligned trades did worse, which usually means the entries were late: by the time positioning agrees, the move is half done. Try entering on the test of the level instead of the confirmation."}`); }
  L.push(`Best: ${best.contract} ${money(best.pnl)}. Worst: ${worst.contract} ${money(worst.pnl)}${worst.why ? ` ("${worst.why}")` : ""}. Reread both entries' notes before your next trade.`);
  return out;
}
async function aiCoach(report){

const HQ = { quotes:new Map(), at:0, loading:false, sector:"all", status:"passes", sort:"chg" };
async function loadHalalQuotes(){
  if (HQ.loading || Date.now() - HQ.at < 5*60e3) return; HQ.loading = true;
  const syms = HALAL_UNIVERSE.filter(r => r[3] !== "fails" && !/ETF/.test(r[2])).map(r => r[0]);
  let got = [];
  try { const r = await fetch(`/api/quotes?symbols=${syms.join(",")}`).catch(()=>null); if (r && r.ok && !(r.headers.get("content-type")||"").includes("text/html")){ const j = await r.json(); got = (j.quotes||[]).filter(q => q && !q.error && q.price); } } catch {}
  if (!got.length){ // static site: Cboe intraday per symbol, in small batches so it doesn't hammer
    for (let i = 0; i < syms.length; i += 8){ const res = await Promise.all(syms.slice(i,i+8).map(tapeFromCboe)); got.push(...res.filter(Boolean)); if (S.tab === "halal") renderTabs(); }
  }
  got.forEach(q => HQ.quotes.set(q.symbol, q)); HQ.at = Date.now(); HQ.loading = false;
  if (S.tab === "halal") renderTabs();
}
