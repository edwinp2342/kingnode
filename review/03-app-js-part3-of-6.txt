function synthBars(spot, iv){
  let seed = 11; const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const mins = {"1m":1,"5m":5,"15m":15,"30m":30,"1h":60,"1d":390}[iv] || 5, perDay = Math.max(1, Math.round(390/mins)), days = iv==="1d" ? 260 : Math.min(10, Math.ceil(400/perDay)+1);
  const bars = []; let px = spot * (1 - 0.02);
  const ymd = dt => dt.toISOString().slice(0,10); const first = new Date(); first.setUTCDate(first.getUTCDate() - Math.round(days*7/5));
  let d = 0, made = 0;
  while (made < days){ const dayD = new Date(first.getTime() + d*86400000); d++; if (dayD.getUTCDay() % 6 === 0) continue; const day = new Date(`${ymd(dayD)}T09:30:00-04:00`); if (day > Date.now()) break; made++;
    for (let i = 0; i < perDay; i++){ const t = day.getTime() + i*mins*60000, o = px, drift = (spot - px) * 0.01, r = (rnd() - 0.5) * spot * 0.0012 * Math.sqrt(mins/5);
      const c = o + drift + r, hi = Math.max(o,c) + rnd()*spot*0.0004*Math.sqrt(mins/5), lo = Math.min(o,c) - rnd()*spot*0.0004*Math.sqrt(mins/5);
      bars.push([t, o, hi, lo, c, Math.round(20000 + rnd()*80000 * (i<6||i>perDay-7 ? 2.2 : 1))]); px = c; } }
  const off = spot - bars[bars.length-1][4], n = bars.length;
  return bars.map((b,i) => { const k = off * (i+1)/n; return [b[0], b[1]+k, b[2]+k, b[3]+k, b[4]+k, b[5]]; });
}
function sma(vals, n){ const out = new Array(vals.length).fill(null); let sum = 0; for (let i = 0; i < vals.length; i++){ sum += vals[i]; if (i >= n) sum -= vals[i-n]; if (i >= n-1) out[i] = sum/n; } return out; }
function bollinger(vals, n=20, k=2){ const m = sma(vals, n); return m.map((mu,i) => { if (mu == null) return null; let v = 0; for (let j = i-n+1; j <= i; j++) v += (vals[j]-mu)**2; const sd = Math.sqrt(v/n); return [mu-k*sd, mu, mu+k*sd]; }); }
function rsi(vals, n=14){ const out = new Array(vals.length).fill(null); let g = 0, l = 0; for (let i = 1; i < vals.length; i++){ const d = vals[i]-vals[i-1]; if (i <= n){ if (d>0) g += d; else l -= d; if (i === n){ g/=n; l/=n; out[i] = l ? 100-100/(1+g/l) : 100; } } else { g = (g*(n-1) + Math.max(d,0))/n; l = (l*(n-1) + Math.max(-d,0))/n; out[i] = l ? 100-100/(1+g/l) : 100; } } return out; }
function isRegular(t){ const et = new Date(new Date(t).toLocaleString("en-US",{timeZone:"America/New_York"})); const m = et.getHours()*60+et.getMinutes(); return m >= 570 && m < 960; }
function ema(vals, n){ const k = 2/(n+1); const out = new Array(vals.length).fill(null); let e = null; for (let i = 0; i < vals.length; i++){ e = e == null ? vals[i] : vals[i]*k + e*(1-k); if (i >= n-1) out[i] = e; } return out; }
function vwap(bars){ const out = []; let pv = 0, v = 0, day = null; for (const b of bars){ const dd = new Date(b[0]).toDateString(); if (dd !== day){ day = dd; pv = 0; v = 0; } const tp = (b[2]+b[3]+b[4])/3; pv += tp*b[5]; v += b[5]; out.push(v ? pv/v : tp); } return out; }

// ---------- Derived flow: when there's no trade feed, build the tape from volume changes between chain refreshes ----------
const DF_KEY = () => `kingnode:dflow:${S.sym}:${new Date().toISOString().slice(0,10)}`;
function deriveFlow(prev, next){
  if (!prev || prev.symbol !== next.symbol || !prev.rows?.length) return [];
  const key = r => `${r[0]}|${r[1]}|${r[2]}`; const pm = new Map(prev.rows.map(r => [key(r), r]));
  const out = []; const now = Date.now();
  for (const r of next.rows){ const p = pm.get(key(r)); if (!p) continue; const dv = r[4] - p[4]; if (dv <= 0) continue;
    const px = r[10] || (r[8]+r[9])/2; if (!px) continue;
    const bid = p[8], ask = p[9]; const side = ask && px >= ask - 0.005 ? "A" : bid && px <= bid + 0.005 ? "B" : "M";
    let t = now; if (r[11]){ const d = Date.parse(String(r[11]).replace(" ","T") + (/[+-]\d\d:?\d\d$|Z$/.test(String(r[11])) ? "" : "-04:00")); if (isFinite(d)) t = d; }
    const prem = Math.round(px*dv*100); const kind = dv >= 500 ? "block" : (dv >= 50 && side !== "M" && (r[9]-r[8]) <= px*0.1) ? "sweep" : "trade";
    out.push({ t, u:next.symbol, o:`${next.symbol}${r[0]}${r[1]}${r[2]}`, exp:r[0], K:r[2], cp:r[1], px:+px.toFixed(2), sz:dv, prem, side, kind, ex:0, derived:true }); }
  return out;
}
function loadDerived(){ try { return JSON.parse(localStorage.getItem(DF_KEY())||"[]"); } catch { return []; } }
function saveDerived(list){ try { localStorage.setItem(DF_KEY(), JSON.stringify(list.slice(-3000))); } catch {} }
function recordDerived(prev, next){
  const fresh = deriveFlow(prev, next); if (!fresh.length) return 0;
  const list = loadDerived().concat(fresh); saveDerived(list); S.derivedCount = list.length;
  if (S.flow && S.flow.derived){ S.flow = { at:Date.now(), live:true, sym:S.sym, derived:true, trades:list }; if (["flow","cflow","play"].includes(S.tab)) renderTabs(); checkFlowAlerts(); }
  return fresh.length;
}

function synthFlow(){
  const c = S.calc; if (!c) return [];
  let seed = 5; const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const out = []; const now = Date.now(); const step = c.spot > 2000 ? 25 : c.spot > 500 ? 5 : c.spot > 200 ? 2.5 : 1;
  for (let i = 0; i < 120; i++){
    const cp = rnd() < 0.55 ? "C" : "P", K = Math.round((c.spot * (1 + (rnd()-0.5)*0.03)) / step) * step, exp = (rnd() < 0.6 ? S.exps[Math.floor(rnd()*Math.min(3,S.exps.length))] : S.allExps[Math.floor(rnd()*S.allExps.length)]) || S.exps[0];
    const ct = c.contracts.find(x => x.exp===exp && x.K===K && x.type===cp); const px = ct ? Math.max(0.05, ct.mid * (0.9 + rnd()*0.2)) : 5 + rnd()*20;
    const sz = rnd() < 0.15 ? 500 + Math.floor(rnd()*3000) : 20 + Math.floor(rnd()*400); const kind = rnd() < 0.25 ? "sweep" : sz >= 500 ? "block" : "trade";
    out.push({ t: now - Math.floor(rnd()*5.5*3600e3), u:S.sym, exp, K, cp, px:+px.toFixed(2), sz, prem: Math.round(px*sz*100), side: rnd() < 0.45 ? "A" : rnd() < 0.8 ? "B" : "M", kind, ex: 300 + Math.floor(rnd()*20) });
  }
  return out.sort((a,b)=>a.t-b.t);
}

  const nodes = lastCol.map((v,ri)=>({v,ri,K:rowK[ri]})).sort((a,b)=>Math.abs(b.v)-Math.abs(a.v));
  const king = nodes.find(n => n.v > 0), short = nodes.find(n => n.v < 0);
  const above = nodes.filter(n => n.v > 0 && n.K > spotNow && n.v > pmax*0.15).sort((a,b)=>a.K-b.K)[0], below = nodes.filter(n => n.v > 0 && n.K < spotNow && n.v > pmax*0.15).sort((a,b)=>b.K-a.K)[0];
  // ---- layout ----
  const ladderW = 300, availW = Math.max(700, (SK.full ? window.innerWidth : (document.getElementById("panel")?.clientWidth || 900)) - 28);
  const L = 66, T = 14, R = 12, B = 30, W = availW - ladderW - 12, plotW = W - L - R, rh = Math.max(11, Math.min(SK.full ? 20 : 15, Math.floor((SK.full ? window.innerHeight - 220 : 540)/rows))), plotH = rows*rh, H = T + plotH + B, dpr = Math.min(2, window.devicePixelRatio||1);
  const wrap = h("div",{style:{position:"relative", width:W+"px", height:H+"px", flex:"none"}});
  const cv = h("canvas",{width:W*dpr, height:H*dpr, style:{width:W+"px", height:H+"px", display:"block"}}); const g = cv.getContext("2d"); g.scale(dpr,dpr);
  const ov = h("canvas",{width:W*dpr, height:H*dpr, style:{width:W+"px", height:H+"px", display:"block", position:"absolute", left:0, top:0, cursor:"crosshair"}}); const og2 = ov.getContext("2d"); og2.scale(dpr,dpr);
  const css = n => getComputedStyle(document.documentElement).getPropertyValue(n).trim();
  g.fillStyle = css("--bg"); g.fillRect(0,0,W,H);
  const off = document.createElement("canvas"); off.width = cols; off.height = rows; const ogx = off.getContext("2d"); const img = ogx.createImageData(cols, rows);
  for (let ci = 0; ci < cols; ci++) for (let ri = 0; ri < rows; ri++){ const [r,gg,b] = seekerColor(sm[ci][ri]/vmax); const p = (ri*cols + ci)*4; img.data[p]=r; img.data[p+1]=gg; img.data[p+2]=b; img.data[p+3]=255; }
  ogx.putImageData(img, 0, 0); g.imageSmoothingEnabled = true; g.imageSmoothingQuality = "high"; g.drawImage(off, 0, 0, cols, rows, L, T, plotW, plotH);
  const X = ci => L + (ci + 0.5)/cols*plotW, yOfK = K => { const i = rowK.findIndex(k => k <= K); const k0 = rowK[i] ?? rowK.at(-1), k1 = rowK[i-1] ?? k0; const f = k1===k0 ? 0.5 : (K-k0)/(k1-k0); return T + (Math.max(0,i) - f + 0.5)*rh; };
  // strike labels (sparse) + faint gridlines
  g.font = "11px " + css("--mono"); g.textBaseline = "middle"; g.textAlign = "right"; const every = Math.max(1, Math.ceil(rows/16));
  rowK.forEach((K,ri) => { if (ri % every) return; const y = T + ri*rh + rh/2; g.fillStyle = css("--muted"); g.fillText(fmt(K), L-8, y); g.strokeStyle = "rgba(255,255,255,.05)"; g.beginPath(); g.moveTo(L, y); g.lineTo(L+plotW, y); g.stroke(); });
  // time axis in ET
  g.textAlign = "center"; g.textBaseline = "alphabetic"; let lastLbl = -Infinity;
  hist.forEach((r,ci) => { const min = Number(new Intl.DateTimeFormat("en-US",{minute:"numeric",timeZone:"America/New_York"}).format(new Date(r.t))); if ((min === 0 || min === 30) && ci - lastLbl >= cols/12){ lastLbl = ci; g.strokeStyle = "rgba(255,255,255,.07)"; g.beginPath(); g.moveTo(X(ci), T); g.lineTo(X(ci), T+plotH); g.stroke(); g.fillStyle = css("--muted"); g.fillText(etTime(r.t), X(ci), H-10); } });
  // key levels: thin dashed lines, no text on the map — the ladder carries the words
  const line = (K, col, dash) => { if (!K || K > rowK[0] || K < rowK.at(-1)) return; const y = yOfK(K); g.strokeStyle = col; g.setLineDash(dash); g.lineWidth = 1; g.beginPath(); g.moveTo(L,y); g.lineTo(L+plotW,y); g.stroke(); g.setLineDash([]); };
  line(c.callWall?.K, "rgba(94,234,212,.7)", [5,4]); line(c.putWall?.K, "rgba(252,165,165,.7)", [5,4]); line(c.flip, "rgba(200,208,220,.6)", [2,3]);
  if (king){ const y = T + king.ri*rh; g.strokeStyle = "rgba(255,255,255,.85)"; g.lineWidth = 1.5; g.strokeRect(L+0.5, y+0.5, plotW-1, rh-1); g.fillStyle = "#fff"; g.font = "700 11px " + css("--sans"); g.textAlign = "left"; g.textBaseline = "middle"; g.fillText("♛", L+6, y+rh/2); }
  // price
  g.lineJoin = "round"; g.lineCap = "round"; const path = () => { g.beginPath(); hist.forEach((r,ci) => { const y = yOfK(r.spot); ci ? g.lineTo(X(ci),y) : g.moveTo(X(ci),y); }); };
  g.strokeStyle = "rgba(0,0,0,.6)"; g.lineWidth = 5; path(); g.stroke(); g.strokeStyle = "#FFFFFF"; g.lineWidth = 2; path(); g.stroke();
  const lastY = yOfK(spotNow); g.fillStyle = "#FFFFFF"; g.beginPath(); g.arc(X(cols-1), lastY, 4, 0, Math.PI*2); g.fill();
  { const txt = fmt(spotNow, 2); g.font = "700 11px " + css("--mono"); const w = g.measureText(txt).width + 10; g.fillStyle = "#FFFFFF"; roundRect(g, L+plotW-w-4, lastY-9, w, 18, 3); g.fill(); g.fillStyle = "#131722"; g.textAlign = "left"; g.textBaseline = "middle"; g.fillText(txt, L+plotW-w+1, lastY); }
  // ---- overlay ----
  const tip = document.getElementById("tip");
  const drawOverlay = ci => { og2.clearRect(0,0,W,H); if (ci == null) return; const x = X(ci), r = hist[ci]; og2.strokeStyle = "rgba(255,255,255,.55)"; og2.lineWidth = 1; og2.setLineDash([3,3]); og2.beginPath(); og2.moveTo(x, T); og2.lineTo(x, T+plotH); og2.stroke(); og2.setLineDash([]); og2.fillStyle = "#fff"; og2.beginPath(); og2.arc(x, yOfK(r.spot), 3.5, 0, Math.PI*2); og2.fill();
    const txt = `${etTime(r.t)} · ${fmt(r.spot,2)}`; og2.font = "600 11px " + css("--mono"); const w = og2.measureText(txt).width + 12; const bx = Math.min(L+plotW-w, Math.max(L, x - w/2)); og2.fillStyle = "rgba(19,23,34,.95)"; og2.strokeStyle = "rgba(255,255,255,.3)"; roundRect(og2, bx, T+plotH+4, w, 18, 4); og2.fill(); og2.stroke(); og2.fillStyle = "#fff"; og2.textAlign = "left"; og2.textBaseline = "middle"; og2.fillText(txt, bx+6, T+plotH+13); ladder(ci); };
  const locate = ev => { const rct = ov.getBoundingClientRect(); const x = ev.clientX-rct.left, y = ev.clientY-rct.top; const ci = Math.floor((x-L)/plotW*cols), ri = Math.floor((y-T)/rh); return {ci, ri, inside: ci>=0 && ci<cols && ri>=0 && ri<rows}; };
  ov.addEventListener("mousemove", ev => { const p = locate(ev); if (!p.inside){ tip.style.display="none"; drawOverlay(SK.scrub); return; } drawOverlay(p.ci);
    const row = hist[p.ci], K = rowK[p.ri], m = new Map(row.k.map(v=>[v[0],v])); const xs = bins[p.ri].map(k => m.get(k)).filter(Boolean); const sum = i => xs.reduce((a,v)=>a+v[i],0);
    tip.style.display="block"; tip.style.left=(ev.clientX+14)+"px"; tip.style.top=(ev.clientY+14)+"px"; tip.textContent = `${fmt(K)} at ${etTime(row.t)}\n${sum(1) >= 0 ? "dealers long" : "dealers short"} ${money(sum(1)*1e5)}${isDelta ? `  (Δ ${money(grid[p.ci][p.ri]*1e5)})` : ""}\nVEX ${money(sum(2)*1e4)}  DEX ${money(sum(3)*1e6)}`; });
  ov.addEventListener("mouseleave", () => { tip.style.display="none"; drawOverlay(SK.scrub); });
  ov.addEventListener("click", ev => { const p = locate(ev); SK.scrub = p.inside && SK.scrub !== p.ci ? p.ci : null; drawOverlay(SK.scrub); });
  wrap.append(cv, ov);
  // replay: step through the session, ladder follows
  const replayBtn = h("button",{class:"btn small", onclick:()=>{ if (SK.timer){ clearInterval(SK.timer); SK.timer = null; replayBtn.textContent = "▶ Replay"; return; } let i = 0; replayBtn.textContent = "■ Stop"; SK.timer = setInterval(() => { SK.scrub = i; drawOverlay(i); i++; if (i >= cols){ clearInterval(SK.timer); SK.timer = null; replayBtn.textContent = "▶ Replay"; } }, 120); }}, "▶ Replay");
  box.querySelector(".sk-head")?.append(replayBtn);
  // ---- ladder (HTML): prices top to bottom, one line each, with the action ----
  const ladderEl = h("div",{class:"ladder"});
  const ladder = ci => {
    const col = ci == null ? lastCol : grid[ci], sp = ci == null ? spotNow : hist[ci].spot, t = ci == null ? "now" : etTime(hist[ci].t);
    const rowsL = [];
    const push = (K, name, kind, size) => { if (!K) return; if (rowsL.some(r => r.K === K)) { const r = rowsL.find(r => r.K === K); r.name += " · " + name; return; } rowsL.push({K, name, kind, size}); };
    const sizeAt = K => { const ri = rowK.findIndex(k => k <= K); return ri >= 0 ? col[ri]*unit : null; };
    push(c.callWall?.K, "Call wall", "sell", sizeAt(c.callWall?.K)); if (above) push(above.K, "Long node above", "sell", above.v*unit);
    if (king) push(king.K, "King node", king.K > sp ? "sell" : "buy", king.v*unit);
    push(sp, "Spot", "spot", null);
    push(c.flip, "Gamma flip", sp > c.flip ? "line" : "reclaim", sizeAt(c.flip));
    if (below) push(below.K, "Long node below", "buy", below.v*unit); push(c.putWall?.K, "Put wall", "buy", sizeAt(c.putWall?.K));
    if (short) push(short.K, "Short node", "avoid", short.v*unit);
    rowsL.forEach(r => { if (r.kind !== "spot" && r.size != null && r.size < 0 && !/flip/i.test(r.name)) r.kind = "avoid"; });
    rowsL.sort((a,b)=>b.K-a.K);
    const verb = {sell:["SELL / TP","#FCA5A5"], buy:["BUY DIP","#5EEAD4"], spot:["NOW","#FFFFFF"], line:["HOLD ABOVE","#C8D0DC"], reclaim:["RECLAIM","#C8D0DC"], avoid:["DON'T FADE","#FCA5A5"]};
    ladderEl.replaceChildren(h("div",{class:"ladder-head"}, h("span",null,"Price ladder"), h("span",{class:"hint"}, t)),
      ...rowsL.map(r => { const [v,col] = verb[r.kind]; return h("div",{class:"lrow"+(r.kind==="spot"?" spot":""), onclick:()=>{ if (r.kind !== "spot"){ S.inspect = {exp:S.exps[0], K:r.K}; } }},
        h("span",{class:"num lpx"}, fmt(r.K, r.kind==="spot"?2:0)), h("span",{class:"lname"}, r.name, r.size != null ? h("small",null, " " + (r.size>=0?"long ":"short ") + money(Math.abs(r.size))) : null), h("span",{class:"lverb", style:{color:col}}, v)); }),
      h("div",{class:"ladder-foot"}, c.em[0] ? `Expected range ${fmt(c.spot - c.em[0].move)} – ${fmt(c.spot + c.em[0].move)}` : "", h("br"), c.total > 0 ? "Positive gamma: dips get bought, rallies get sold." : "Negative gamma: moves extend — trade with the break, not against it."));
  };
  ladder(null); if (SK.scrub != null && SK.scrub < cols) drawOverlay(SK.scrub);
  box.append(h("div",{class:"sk-body"}, h("div",{class:"heat", style:{padding:"4px", overflow:"auto", flex:"none"}}, wrap), ladderEl));
  // legend under the map, one line
  box.append(h("div",{class:"sk-legend"}, h("span",null, h("i",{style:{background:"#78C4AA"}}), isDelta ? "strengthening" : "dealers long — support, pins"), h("span",null, h("i",{style:{background:"#AA8CD2"}}), isDelta ? "unwinding" : "dealers short — accelerates"), h("span",null, h("i",{style:{background:"#fff"}}), "price"), h("span",null,"♛ king node"), h("span",{class:"hint"},"hover for size · click a column to pin a time · times in ET")));
  return box;
}
