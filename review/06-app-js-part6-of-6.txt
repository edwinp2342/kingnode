function flowSetups(){
  const c = S.calc, tr = (S.flow?.trades||[]).filter(t => t.u === S.sym), pb = getPb();
  const agg = new Map();
  for (const t of tr){ const k = `${t.exp}|${t.K}|${t.cp}`; const a = agg.get(k) || {exp:t.exp, K:t.K, cp:t.cp, prem:0, n:0, ask:0, sweeps:0, blocks:0, last:0, sz:0}; a.prem += t.prem; a.n++; a.sz += t.sz; if (t.side==="A") a.ask += t.prem; if (t.kind==="sweep") a.sweeps++; if (t.kind==="block") a.blocks++; a.last = Math.max(a.last, t.t); agg.set(k,a); }
  const out = [];
  for (const a of agg.values()){
    const d = dte(a.exp); if (d < pb.minDte) continue;                       // 30+ DTE only — that's the rule
    const ct = c.allContracts.find(x => x.exp===a.exp && x.K===a.K && x.type===a.cp); const oi = ct?.oi || 0, vol = ct?.vol || 0, volOi = oi ? vol/oi : 0;
    const askPct = a.ask/a.prem; if (askPct < 0.55) continue;                  // paid up, not sold
    const bullish = a.cp === "C";
    const aligned = bullish ? (c.spot > (c.flip||0) && (c.callWall?.K||Infinity) > a.K*0.98) : (c.spot < (c.flip||Infinity) || (c.putWall?.K||0) < a.K);
    const score = Math.round(Math.min(100, a.prem/2e5*20 + askPct*25 + Math.min(20, volOi*5) + a.sweeps*6 + a.blocks*4 + (aligned ? 15 : 0)));
    out.push({...a, dte:d, oi, vol, volOi, askPct, score, aligned, bullish, mid:ct?.mid||0, delta:ct?.delta||0, iv:ct?.iv||0});
  }
  return out.sort((a,b)=>b.score-a.score).slice(0,12);
}

// ---------- Institutions: where the big money is leaning ----------
async function loadInst(){
  const sym = S.sym; if (S.inst && S.inst.sym === sym && Date.now() - S.inst.at < 5*60e3) return;
  const get = async (u) => { try { const r = await fetch(u); if (!r.ok) throw 0; return await r.json(); } catch { return null; } };
  const [sv, ins, daily, intra] = await Promise.all([get(`/api/shortvol?symbol=${sym}`), get(`/api/insiders?symbol=${sym}`), get(`/api/bars?symbol=${sym}&interval=1d&range=6mo`), get(`/api/bars?symbol=${sym}&interval=5m&range=5d`)]);
  if (!S.flow || S.flow.sym !== sym) { try { const r = await fetch(`/api/flow?symbol=${sym}`); const j = r.ok ? await r.json() : null; if (j?.trades?.length){ S.flow = {...j, sym}; S.flowDemo = false; } else { S.flow = {at:Date.now(), live:false, sym, trades:synthFlow()}; S.flowDemo = true; } } catch { S.flow = {at:Date.now(), live:false, sym, trades:synthFlow()}; S.flowDemo = true; } }
  const spot = S.chain?.spot || 100;
  const demo = { sv: !sv?.days?.length, ins: !ins?.filings, daily: !daily?.bars?.length, intra: !intra?.bars?.length };
  S.inst = { sym, at: Date.now(), demo,
    sv: sv?.days?.length ? sv.days : synthShort(spot),
    ins: ins?.filings ? ins : null,
    daily: daily?.bars?.length ? daily.bars : synthBars(spot, "1d"),
    intra: intra?.bars?.length ? intra.bars : synthBars(spot, "5m") };
  if (S.tab === "inst") renderTabs();
