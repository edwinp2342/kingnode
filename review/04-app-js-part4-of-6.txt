function roundRect(g, x, y, w, h, r){ g.beginPath(); g.moveTo(x+r,y); g.lineTo(x+w-r,y); g.quadraticCurveTo(x+w,y,x+w,y+r); g.lineTo(x+w,y+h-r); g.quadraticCurveTo(x+w,y+h,x+w-r,y+h); g.lineTo(x+r,y+h); g.quadraticCurveTo(x,y+h,x,y+h-r); g.lineTo(x,y+r); g.quadraticCurveTo(x,y,x+r,y); g.closePath(); }

const QD = { view:"bull", horizon:10, stop:20, target:60, minPop:30, maxSpread:12, minDte:7, maxDte:60, structures:{call:true,put:true,debit:true,credit:true}, sel:null, mc:new Map(), seed:17 };
const N = x => 0.5*(1+erf(x/Math.SQRT2));
const Nd2 = (S0, K, T, sig) => { const sq = sig*Math.sqrt(Math.max(T,1e-6)); return N((Math.log(S0/K) - 0.5*sig*sig*T)/sq); };
function realizedVol(bars){
  if (!bars || bars.length < 30) return null;
  // group by session → daily closes; if fewer than 12 sessions, fall back to intraday returns scaled to annual
  const byDay = new Map(); for (const b of bars){ byDay.set(new Date(b[0]).toDateString(), b[4]); }
  const closes = [...byDay.values()];
  if (closes.length >= 12){ const r = closes.slice(1).map((c,i)=>Math.log(c/closes[i])); const n = Math.min(20, r.length); const rr = r.slice(-n); const m = rr.reduce((a,b)=>a+b,0)/n; const v = rr.reduce((a,b)=>a+(b-m)**2,0)/(n-1); return {rv:Math.sqrt(v*252), n, basis:`${n} sessions, close-to-close`}; }
  const mins = (bars[1][0]-bars[0][0])/60000 || 5, per = 390/mins; const r = bars.slice(1).map((b,i)=>Math.log(b[4]/bars[i][4])).filter(isFinite); const n = Math.min(400, r.length); const rr = r.slice(-n); const m = rr.reduce((a,b)=>a+b,0)/n; const v = rr.reduce((a,b)=>a+(b-m)**2,0)/(n-1); return {rv:Math.sqrt(v*per*252), n, basis:`${n} intraday bars, scaled`};
}
function volSurface(){
  const c = S.calc, spot = c.spot; const byExp = new Map();
  for (const x of c.allContracts){ if (!x.iv || x.iv <= 0 || x.iv > 5) continue; const e = byExp.get(x.exp) || {exp:x.exp, dte:dte(x.exp), atm:null, atmDist:Infinity, p25:null, c25:null, p25d:1, c25d:1};
    const d = Math.abs(x.K - spot); if (d < e.atmDist){ e.atmDist = d; e.atm = x.iv; }
    if (x.type==="P" && Math.abs(Math.abs(x.delta)-0.25) < e.p25d){ e.p25d = Math.abs(Math.abs(x.delta)-0.25); e.p25 = x.iv; }
    if (x.type==="C" && Math.abs(x.delta-0.25) < e.c25d){ e.c25d = Math.abs(x.delta-0.25); e.c25 = x.iv; }
    byExp.set(x.exp, e); }
  return [...byExp.values()].sort((a,b)=>a.dte-b.dte);
}

function quantCandidates(){
  const c = S.calc, spot = c.spot, bull = QD.view === "bull", bear = QD.view === "bear";
  const cs = c.allContracts.filter(x => { const d = dte(x.exp); return d >= QD.minDte && d <= QD.maxDte && x.oi >= 50 && x.mid > 0.05 && Math.abs(x.K/spot-1) <= 0.15; });
  const byKey = new Map(cs.map(x => [`${x.exp}|${x.type}|${x.K}`, x]));
  const out = [];
  const liq = x => (x.ask - x.bid) / (x.mid || 1);
  const push = o => { if (o.spreadPct <= QD.maxSpread/100) out.push(o); };
  for (const x of cs){
    const T = Math.max(1, dte(x.exp))/365;
    if (QD.structures.call && x.type==="C" && bull && x.K >= spot*0.97 && x.K <= spot*1.08){ push({kind:"Long call", legs:[{...x, side:+1}], debit:x.mid, maxLoss:x.mid, maxGain:Infinity, be:x.K + x.mid, spreadPct:liq(x), exp:x.exp, T, dir:+1, K:x.K}); }
    if (QD.structures.put && x.type==="P" && bear && x.K <= spot*1.03 && x.K >= spot*0.92){ push({kind:"Long put", legs:[{...x, side:+1}], debit:x.mid, maxLoss:x.mid, maxGain:x.K - x.mid, be:x.K - x.mid, spreadPct:liq(x), exp:x.exp, T, dir:-1, K:x.K}); }
    if (QD.structures.debit && bull && x.type==="C" && x.K >= spot*0.98 && x.K <= spot*1.05){ for (const w of [1,2,3]){ const ks = [...new Set(cs.filter(y=>y.exp===x.exp&&y.type==="C").map(y=>y.K))].sort((a,b)=>a-b); const i = ks.indexOf(x.K); const K2 = ks[i+w]; const y = K2 && byKey.get(`${x.exp}|C|${K2}`); if (!y) continue; const debit = x.mid - y.mid; if (debit <= 0.05) continue; push({kind:"Call debit spread", legs:[{...x, side:+1},{...y, side:-1}], debit, maxLoss:debit, maxGain:(K2-x.K)-debit, be:x.K+debit, spreadPct:(liq(x)+liq(y))/2, exp:x.exp, T, dir:+1, K:x.K, K2}); } }
    if (QD.structures.debit && bear && x.type==="P" && x.K <= spot*1.02 && x.K >= spot*0.95){ for (const w of [1,2,3]){ const ks = [...new Set(cs.filter(y=>y.exp===x.exp&&y.type==="P").map(y=>y.K))].sort((a,b)=>b-a); const i = ks.indexOf(x.K); const K2 = ks[i+w]; const y = K2 && byKey.get(`${x.exp}|P|${K2}`); if (!y) continue; const debit = x.mid - y.mid; if (debit <= 0.05) continue; push({kind:"Put debit spread", legs:[{...x, side:+1},{...y, side:-1}], debit, maxLoss:debit, maxGain:(x.K-K2)-debit, be:x.K-debit, spreadPct:(liq(x)+liq(y))/2, exp:x.exp, T, dir:-1, K:x.K, K2}); } }
    if (QD.structures.credit && (bull || QD.view==="neutral") && x.type==="P" && x.K <= spot*0.99 && x.K >= spot*0.90){ for (const w of [1,2]){ const ks = [...new Set(cs.filter(y=>y.exp===x.exp&&y.type==="P").map(y=>y.K))].sort((a,b)=>b-a); const i = ks.indexOf(x.K); const K2 = ks[i+w]; const y = K2 && byKey.get(`${x.exp}|P|${K2}`); if (!y) continue; const credit = x.mid - y.mid, width = x.K - K2; if (credit <= 0.05 || credit >= width) continue; push({kind:"Put credit spread", legs:[{...x, side:-1},{...y, side:+1}], debit:-credit, maxLoss:width-credit, maxGain:credit, be:x.K-credit, spreadPct:(liq(x)+liq(y))/2, exp:x.exp, T, dir:+1, K:x.K, K2, credit:true}); } }
    if (QD.structures.credit && (bear || QD.view==="neutral") && x.type==="C" && x.K >= spot*1.01 && x.K <= spot*1.10){ for (const w of [1,2]){ const ks = [...new Set(cs.filter(y=>y.exp===x.exp&&y.type==="C").map(y=>y.K))].sort((a,b)=>a-b); const i = ks.indexOf(x.K); const K2 = ks[i+w]; const y = K2 && byKey.get(`${x.exp}|C|${K2}`); if (!y) continue; const credit = x.mid - y.mid, width = K2 - x.K; if (credit <= 0.05 || credit >= width) continue; push({kind:"Call credit spread", legs:[{...x, side:-1},{...y, side:+1}], debit:-credit, maxLoss:width-credit, maxGain:credit, be:x.K+credit, spreadPct:(liq(x)+liq(y))/2, exp:x.exp, T, dir:-1, K:x.K, K2, credit:true}); } }
  }
  return out;
}
// structure value at a given underlying/time (BS with each leg's own IV)
function structValue(o, S0, tYears, ivShift=0){ return o.legs.reduce((a,l) => a + l.side * bsPrice(S0, l.K, Math.max(0, tYears), Math.max(0.02, (l.iv||0.2) + ivShift), l.type), 0); }
function payoffAtExpiry(o, S0){ return o.legs.reduce((a,l) => a + l.side * Math.max(0, l.type==="C" ? S0-l.K : l.K-S0), 0) - o.debit; }
// analytic metrics under two measures: implied (what the market prices) and realized (what the stock has been doing)
function analytic(o, rv){
  const c = S.calc, S0 = c.spot, T = o.T, ivs = o.legs[0].iv || 0.2;
  const pProfitExp = sig => { // P(payoff at expiry > 0) under lognormal, drift 0
    const sq = sig*Math.sqrt(T); if (o.kind.includes("credit")){ return o.dir > 0 ? Nd2(S0, o.be, T, sig) : 1 - Nd2(S0, o.be, T, sig); } return o.dir > 0 ? Nd2(S0, o.be, T, sig) : 1 - Nd2(S0, o.be, T, sig); };
  const pItm = sig => o.dir > 0 ? Nd2(S0, o.K, T, sig) : 1 - Nd2(S0, o.K, T, sig);
  // expected value at expiry by numeric integration under RV lognormal
  const ev = sig => { const n = 240, sq = sig*Math.sqrt(T); let e = 0; for (let i = 0; i < n; i++){ const z = -4 + 8*(i+0.5)/n; const w = Math.exp(-z*z/2)/Math.sqrt(2*Math.PI)*(8/n); const ST = S0*Math.exp(-0.5*sig*sig*T + sq*z); e += w*payoffAtExpiry(o, ST); } return e; };
  return { pItmIV:pItm(ivs), pItmRV:pItm(rv), popIV:pProfitExp(ivs), popRV:pProfitExp(rv), evIV:ev(ivs), evRV:ev(rv) };
}
// Monte Carlo with your rules: stop at −stop% / target at +target% of the debit, exit at horizon otherwise. Student-t(4) shocks for fat tails.
function monteCarlo(o, rv, paths=3000){
  const c = S.calc, S0 = c.spot, H = Math.min(QD.horizon, dte(o.exp)), steps = H, dt = 1/252, sig = rv;
  let seed = QD.seed + Math.round(o.K*7 + (o.K2||0)*3 + o.legs.length); const rnd = () => (seed = (seed * 48271) % 2147483647) / 2147483647;
  const gauss = () => { const u = Math.max(1e-9, rnd()), v = rnd(); return Math.sqrt(-2*Math.log(u))*Math.cos(2*Math.PI*v); };
  const tdraw = () => { const z = gauss(); const chi = (gauss()**2 + gauss()**2 + gauss()**2 + gauss()**2)/4; return z/Math.sqrt(chi) * Math.sqrt(2/4); }; // t(4) scaled to unit variance
  const risk = o.credit ? o.maxLoss : o.debit; const stopLvl = -QD.stop/100*risk, tgtLvl = QD.target/100*risk;
  const v0 = structValue(o, S0, o.T); // anchor to the model's own t=0 value so a model/market gap doesn't show up as instant P&L
  let hitT = 0, hitS = 0, profH = 0, sumPnl = 0, sumT = 0; const pnls = [];
  for (let p = 0; p < paths; p++){
    let St = S0, pnl = 0, done = false;
    for (let d = 1; d <= steps; d++){
      St *= Math.exp(-0.5*sig*sig*dt + sig*Math.sqrt(dt)*tdraw());
      const val = structValue(o, St, o.T - d*dt); pnl = val - v0;
      if (pnl >= tgtLvl){ hitT++; sumT += d; done = true; break; }
      if (pnl <= stopLvl){ hitS++; sumT += d; done = true; break; }
    }
    if (!done){ sumT += steps; if (pnl > 0) profH++; }
    sumPnl += Math.max(stopLvl, Math.min(tgtLvl, pnl)); pnls.push(pnl);
  }
  pnls.sort((a,b)=>a-b); const q = f => pnls[Math.floor(f*(pnls.length-1))];
  return { pTarget:hitT/paths, pStop:hitS/paths, pProfit:(hitT+profH)/paths, ev:sumPnl/paths, avgDays:sumT/paths, p10:q(.1), p50:q(.5), p90:q(.9), horizon:H, risk };
}
function rankQuant(rvInfo){
  const rv = rvInfo?.rv || ivAt(30)*0.9; const cands = quantCandidates(); const c = S.calc;
  const flowLean = (() => { const tr = (S.flow?.trades||[]).filter(t=>t.u===S.sym); const b = tr.filter(t=>(t.cp==="C"&&t.side==="A")||(t.cp==="P"&&t.side==="B")).reduce((a,t)=>a+t.prem,0), s = tr.filter(t=>(t.cp==="P"&&t.side==="A")||(t.cp==="C"&&t.side==="B")).reduce((a,t)=>a+t.prem,0); return (b-s)/((b+s)||1); })();
  const rows = cands.map(o => { const a = analytic(o, rv); const risk = o.credit ? o.maxLoss : o.debit; const evr = a.evRV/risk;
    const posAlign = o.dir > 0 ? (c.spot > (c.flip||0) ? 1 : 0) + (c.callWall && o.K <= c.callWall.K ? 1 : 0) : (c.spot < (c.flip||Infinity) ? 1 : 0) + (c.putWall && o.K >= c.putWall.K ? 1 : 0);
    const edge = a.evRV - a.evIV; // what realized vol says vs what implied prices → the mispricing a desk harvests
    const score = evr*40 + a.popRV*30 + posAlign*6 + (o.dir*flowLean)*8 - o.spreadPct*60 + Math.min(6, Math.log10(1+(o.legs[0].oi||0)));
    return {...o, ...a, risk, evr, edge, posAlign, flowLean, score, rv}; })
    .filter(r => r.popRV*100 >= QD.minPop).sort((a,b)=>b.score-a.score);
  return rows.slice(0, 40);
}

function quantCard(o, rv){
  const c = S.calc, S0 = c.spot; const key = `${o.kind}|${o.exp}|${o.K}|${o.K2||""}|${QD.horizon}|${QD.stop}|${QD.target}`;
  let mc = QD.mc.get(key); if (!mc){ mc = monteCarlo(o, rv || ivAt(30)*0.9); QD.mc.set(key, mc); }
  const pb = getPb(), risk$ = pb.account*pb.maxRiskPct/100, perContract = o.risk*100; const stopPerContract = perContract*QD.stop/100;
  const qty = Math.max(1, Math.floor(risk$ / stopPerContract));
  const variance = Math.max(1e-6, ((mc.p90 - mc.p10)/2.56)**2); const kelly = Math.max(0, Math.min(0.25, (mc.ev/perContract*100) / (variance/(perContract*perContract/10000)) )); // crude; halved below
  const halfKelly = kelly/2; const kellyQty = Math.max(0, Math.floor(pb.account*halfKelly / perContract));
  // payoff diagram
  const W = 760, H = 260, pad = {l:56,r:16,t:16,b:28}; const lo = S0*0.85, hi = S0*1.15; const xs = []; for (let i = 0; i <= 120; i++) xs.push(lo + (hi-lo)*i/120);
  const v0 = structValue(o, S0, o.T);
  const yExp = xs.map(x => payoffAtExpiry(o, x)), yNow = xs.map(x => structValue(o, x, Math.max(0, o.T - Math.min(QD.horizon, dte(o.exp))/365)) - v0);
  const yMin = Math.min(...yExp, ...yNow), yMax = Math.max(...yExp.filter(isFinite), ...yNow);
  const X = x => pad.l + (x-lo)/(hi-lo)*(W-pad.l-pad.r), Y = y => H-pad.b - (y-yMin)/((yMax-yMin)||1)*(H-pad.t-pad.b);
  const svg = svgEl("svg",{viewBox:`0 0 ${W} ${H}`, width:"100%", style:"display:block"});

  // profit/loss fill at expiry
  const area = xs.map((x,i)=>`${X(x)},${Y(yExp[i])}`).join(" "); svg.append(svgEl("polyline",{points:area, fill:"none", stroke:"#5EEAD4", "stroke-width":2}));

  const vline = (x, col, lbl) => { if (x < lo || x > hi) return; svg.append(svgEl("line",{x1:X(x),y1:pad.t,x2:X(x),y2:H-pad.b,stroke:col,"stroke-dasharray":"3 3"})); const t = svgEl("text",{x:X(x)+3,y:pad.t+11}); t.textContent = lbl; t.style.fill = col; svg.append(t); };
  vline(S0, "#fff", `spot ${fmt(S0,2)}`); vline(o.be, "#F97066", `BE ${fmt(o.be,2)}`); if (c.callWall) vline(c.callWall.K, "rgba(94,234,212,.7)", "CW"); if (c.putWall) vline(c.putWall.K, "rgba(252,165,165,.7)", "PW");

  const sig = rv || ivAt(30); const Hd = Math.min(QD.horizon, dte(o.exp)); const W2 = 760, H2 = 220, p2 = {l:56,r:16,t:14,b:26};

  const bands = [[1,"rgba(45,212,191,.18)"],[2,"rgba(45,212,191,.08)"]]; const yLo = S0*Math.exp(-2*sig*Math.sqrt(Hd/252))*0.99, yHi = S0*Math.exp(2*sig*Math.sqrt(Hd/252))*1.01;
  const X2 = d => p2.l + d/Hd*(W2-p2.l-p2.r), Y2 = v => H2-p2.b - (v-yLo)/((yHi-yLo)||1)*(H2-p2.t-p2.b);
  for (const [k,col] of bands.reverse()){ const up = [], dn = []; for (let d = 0; d <= Hd; d++){ const s = sig*Math.sqrt(d/252); up.push(`${X2(d)},${Y2(S0*Math.exp(k*s))}`); dn.unshift(`${X2(d)},${Y2(S0*Math.exp(-k*s))}`); } cone.append(svgEl("polygon",{points:[...up,...dn].join(" "), fill:col})); }

  const ivShiftBox = h("div",{class:"hint"},"");

      h("div",null, h("h3",{class:"qh3"},"Payoff — expiry (solid) vs at +" + Hd + "d (dashed)"), svg),

    h("div",{style:{marginTop:"12px",display:"flex",gap:"12px",alignItems:"center",flexWrap:"wrap"}}, h("span",{class:"hint"},"What if IV moves?"), shiftInput, ivShiftBox),

function quantNarrative(o, mc, rv){
  const c = S.calc; const cheap = o.edge > 0; const parts = [];
  parts.push(`${o.kind} on ${S.sym}: the market prices a ${fmt(o.pItmIV*100,0)}% chance of finishing in the money; the stock's realized vol says ${fmt(o.pItmRV*100,0)}%.`);
  parts.push(cheap ? `Implied is below realized here, so you're paying less than the movement justifies — that's the kind of mispricing a desk buys.` : `Implied is above realized, so you're paying for more movement than the stock has delivered; the edge is on the seller's side unless a catalyst changes the vol.`);
  parts.push(`With your rules (stop ${QD.stop}%, target ${QD.target}%, ${mc.horizon}-day horizon) the simulation hits the target first ${fmt(mc.pTarget*100,0)}% of the time and the stop first ${fmt(mc.pStop*100,0)}%, for an expected ${mc.ev>=0?"gain":"loss"} of ${fmt(Math.abs(mc.ev),2)} per contract.`);
  if (o.posAlign >= 2) parts.push(`Positioning agrees: spot is on the right side of the flip and the strike sits inside the wall.`); else if (o.posAlign === 0) parts.push(`Positioning disagrees: dealers are leaning against this direction at these strikes, so expect the move to be ground out, not gapped.`);
  if (o.spreadPct > 0.08) parts.push(`Watch the fill — the market is ${fmt(o.spreadPct*100,0)}% wide, which is a real tax on a trade this size.`);
  return parts.join(" ");
}

function emaArr(v, n){ return ema(v, n); }
function analyzeTicker(sym, chain, daily){
  const exps = [...new Set(chain.rows.map(r=>r[0]))].sort().slice(0, 6);
  const calc = computeFrom({...chain, symbol:sym}, exps, 8, S.set.model); calc.symbol = sym;
  const spot = calc.spot; const closes = daily ? daily.map(b=>b[4]) : [];
  const e21 = closes.length > 25 ? emaArr(closes,21).at(-1) : null, e50 = closes.length > 55 ? emaArr(closes,50).at(-1) : null;
  const r14 = closes.length > 16 ? rsi(closes,14).at(-1) : null;
  const chg1 = closes.length > 1 ? closes.at(-1)/closes.at(-2)-1 : (chain.changePct||0)/100, chg5 = closes.length > 6 ? closes.at(-1)/closes.at(-6)-1 : null, chg20 = closes.length > 21 ? closes.at(-1)/closes.at(-21)-1 : null;
  const rvInfo = daily && daily.length > 12 ? realizedVol(daily) : null; const rv = rvInfo?.rv || null;
  const iv30 = withCalc(calc, daily, () => ivAt(30)); const vrp = rv != null ? iv30 - rv : null;
  const aboveFlip = calc.flip ? spot > calc.flip : null; const posG = calc.total > 0;
  const cw = calc.callWall?.K, pw = calc.putWall?.K; const toCW = cw ? cw/spot-1 : null, toPW = pw ? 1-pw/spot : null;
  const trendUp = e21 && e50 ? (spot > e21 && e21 > e50) : (chg20 != null ? chg20 > 0 : null);
  const trendDown = e21 && e50 ? (spot < e21 && e21 < e50) : (chg20 != null ? chg20 < 0 : null);
  const pullback = e21 ? Math.abs(spot/e21-1) < 0.015 : false; const extended = e21 ? spot/e21-1 > 0.04 : chg5 != null && chg5 > 0.06;
  const reasons = []; let signal = "Watch", score = 0, view = "neutral";
  // --- equity rules ---
  if (trendUp){ score += 25; reasons.push("uptrend (price > 21 EMA > 50 EMA)"); view = "bull"; }
  if (trendDown){ score -= 25; reasons.push("downtrend (price < 21 EMA < 50 EMA)"); view = "bear"; }
  if (aboveFlip === true){ score += 10; reasons.push(`above gamma flip ${fmt(calc.flip)}`); } else if (aboveFlip === false){ score -= 10; reasons.push(`below gamma flip ${fmt(calc.flip)}`); }
  if (posG){ score += 5; reasons.push("dealers long gamma — dips bought"); } else { reasons.push("dealers short gamma — moves extend"); }
  if (toPW != null && toPW < 0.02 && toPW >= 0){ score += 12; reasons.push(`put wall ${fmt(pw)} within ${fmt(toPW*100,1)}% below — support under you`); }
  if (toCW != null && toCW < 0.01 && toCW >= 0){ score -= 10; reasons.push(`at the call wall ${fmt(cw)} — resistance`); }
  if (pullback && trendUp){ score += 12; reasons.push("pullback to the 21 EMA in an uptrend"); }
  if (extended){ score -= 12; reasons.push("extended above the 21 EMA — chase risk"); }
  if (r14 != null){ if (r14 < 35){ score += 8; reasons.push(`RSI ${fmt(r14,0)} — washed out`); } else if (r14 > 72){ score -= 10; reasons.push(`RSI ${fmt(r14,0)} — overbought`); } }
  if (chg1 > 0.04){ score -= 6; reasons.push(`+${fmt(chg1*100,1)}% today — don't chase`); } if (chg1 < -0.04 && trendUp){ score += 6; reasons.push(`${fmt(chg1*100,1)}% today inside an uptrend — buy-low candidate`); }
  if (vrp != null){ if (vrp > 0.04) reasons.push(`options rich (IV ${fmt(iv30*100,0)}% vs RV ${fmt(rv*100,0)}%) — prefer spreads / shares`); else if (vrp < -0.02) reasons.push(`options cheap (IV ${fmt(iv30*100,0)}% vs RV ${fmt(rv*100,0)}%) — long premium works`); }
  if (score >= 30) signal = "Buy"; else if (score >= 15) signal = "Accumulate"; else if (score <= -25) signal = "Avoid / short"; else if (score <= -10) signal = "Trim";
  if (signal === "Watch" && view === "neutral") view = chg20 != null && chg20 > 0 ? "bull" : "bear";
  // --- equity plan ---
  const entry = signal === "Buy" || signal === "Accumulate" ? (pullback || !e21 ? spot : Math.max(e21, pw||0)) : null;
  const stop = entry ? (pw && pw < entry ? Math.min(pw*0.995, e50 || pw*0.995) : (e50 || entry*0.95)) : null;
  const target = entry ? (cw && cw > entry*1.01 ? cw : entry*1.06) : null;
  // --- option structure from the quant engine ---
  let best = null;
  try { best = withCalc(calc, daily, () => { const saveView = QD.view; QD.view = view === "bear" ? "bear" : "bull"; const rows = rankQuant(rvInfo); QD.view = saveView; return rows[0] || null; }); } catch {}
  return { sym, name:NAMES.get(sym)||"", spot, chg1, chg5, chg20, rsi:r14, e21, e50, trendUp, trendDown, aboveFlip, posG, cw, pw, flip:calc.flip, iv30, rv, vrp, score, signal, view, reasons, entry, stop, target, best, halal:halalStatus(sym), at:Date.now() };
}
async function runIdeas(){
  if (ID.running) return; ID.running = true; const syms = IDEA_UNIVERSES[ID.universe]().filter(s => !ID.halal || halalStatus(s).s !== "excluded");
  let done = 0;
  for (const sym of syms){
    const cached = ID.results.get(sym); if (cached && Date.now() - cached.at < 10*60e3){ done++; continue; }
    ID.progress = `${done+1}/${syms.length} · ${sym}`; if (S.tab === "ideas") renderTabs();
    const [chain, daily] = await Promise.all([fetchChainAny(sym), fetchBarsAny(sym)]);
    if (chain){ try { ID.results.set(sym, analyzeTicker(sym, chain, daily)); } catch(e){ console.warn("ideas", sym, e); } }
    else if (S.demo){ try { const demo = synthChain(sym); ID.results.set(sym, {...analyzeTicker(sym, demo, synthBars(demo.spot,"1d")), demo:true}); } catch(e){ console.error("ideas demo", sym, e.message, e.stack); } }
    done++;
  }
  ID.running = false; ID.progress = ""; ID.at = Date.now(); if (S.tab === "ideas") renderTabs();
}
