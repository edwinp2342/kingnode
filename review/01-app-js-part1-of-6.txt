function h(tag, attrs, ...kids){
  const el = document.createElement(tag);
  if (attrs) for (const [k,v] of Object.entries(attrs)){
    if (v == null || v === false) continue;
    if (k === "class") el.className = v; else if (k.startsWith("on")) el.addEventListener(k.slice(2), v);
    else if (k === "style" && typeof v === "object") Object.assign(el.style, v);
    else if (k in el && k !== "list" && k !== "form") { try { el[k] = v; } catch { el.setAttribute(k, v); } } else el.setAttribute(k, v === true ? "" : v);
  }
  for (const k of kids.flat()) if (k != null && k !== false) el.append(k.nodeType ? k : String(k));
  return el;
}

function synthChain(sym){
  // Deterministic demo chain so the terminal is usable without a data feed (clearly labeled).
  // Demo seeds — approximate late-September 2026 levels so the demo looks like today's tape. Live data replaces all of this.
  const spot = {SPX:7743.41, SPY:772.4, QQQ:642, IWM:282, NVDA:221, TSLA:407, AAPL:325, AMZN:255, MSFT:545, META:580}[sym] || 100;
  const step = spot > 2000 ? 25 : spot > 500 ? 5 : spot > 200 ? 2.5 : 1;
  const rows = []; const today = new Date();
  const exps = []; for (let i = 0; exps.length < 8; i++){ const d = new Date(today.getTime() + i*86400000); if (isWeekday(d) && (i===0 || d.getDay()===5 || exps.length<3)) exps.push(d.toISOString().slice(0,10)); }
  for (let m = 1; m <= 3; m++){ const d = new Date(today.getFullYear(), today.getMonth()+m, 15); while (d.getDay() !== 5) d.setDate(d.getDate()+1); exps.push(d.toISOString().slice(0,10)); }
  let seed = 7; const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  exps.forEach((exp, ei) => {
    const T = Math.max(0.5, dte(exp)) / 365, iv0 = 0.14 + ei*0.004;
    for (let k = -30; k <= 30; k++){
      const K = Math.round((spot + k*step) / step) * step;
      const m = Math.log(K/spot), iv = iv0 + Math.max(0, -m)*0.9 + Math.abs(m)*0.25;
      const d1 = (Math.log(spot/K) + (iv*iv/2)*T) / (iv*Math.sqrt(T)), nd = Math.exp(-d1*d1/2)/Math.sqrt(2*Math.PI);
      const gamma = nd / (spot*iv*Math.sqrt(T));
      const cdf = x => 0.5*(1+erf(x/Math.SQRT2)), delta = cdf(d1);
      const wall = Math.abs(K - Math.round(spot*1.02/step)*step) < 1 ? 14 : Math.abs(K - Math.round(spot*0.97/step)*step) < 1 ? 12 : 1;
      const base = Math.exp(-Math.abs(k)/10) * (ei===0 ? 30000 : 14000/(ei+1));
      const cOI = Math.round(base * (K>spot ? wall : 1) * (0.6 + rnd())), pOI = Math.round(base * (K<spot ? wall : 1) * (0.6 + rnd()));
      const cMid = Math.max(0.05, spot*nd*iv*Math.sqrt(T)*0.8 + Math.max(0, spot-K)), pMid = Math.max(0.05, cMid - (spot-K));
      rows.push([exp,"C",K,cOI,Math.round(cOI*(0.1+rnd()*(k===2?4:1))),iv,gamma,delta,cMid*0.98,cMid*1.02,cMid]);
      rows.push([exp,"P",K,pOI,Math.round(pOI*(0.1+rnd()*(k===-3?4:1))),iv,gamma,delta-1,pMid*0.98,pMid*1.02,pMid]);
    }
  });
  return { symbol:sym, spot, asOf:new Date().toISOString(), delayed:true, change:spot*0.0051, changePct:0.51, rows };
}
function erf(x){ const s = Math.sign(x); x = Math.abs(x); const t = 1/(1+0.3275911*x); const y = 1-(((((1.061405429*t-1.453152027)*t)+1.421413741)*t-0.284496736)*t+0.254829592)*t*Math.exp(-x*x); return s*y; }

// ---------- math ----------
function compute(){ S.calc = computeFrom(S.chain, S.exps, S.set.range, S.set.model); recordHistory(); recordTracked(); recordSeeker(); trackLevelTests(); }
function computeFrom(d, expsSel, range, model){
  const spot = d.spot, mult = 100 * spot * spot * 0.01; // $ gamma per 1% move
  const byCell = new Map(), byStrike = new Map(), byExp = new Map();
  const hi = spot * (1 + range/100), lo = spot * (1 - range/100);
  const contracts = [], allContracts = [];
  for (const r of d.rows){
    const [exp, type, K, oi, vol, iv, gamma, delta, bid, ask, last] = r;
    allContracts.push({exp, type, K, oi, vol, iv, delta, bid, ask, mid:(bid+ask)/2 || last});
    if (!expsSel.includes(exp)) continue;
    // Standard dealer model: dealers are long calls (positive gamma) and short puts (negative gamma).
    // "Net short" model treats dealers as short everything customers bought: both signs negative → magnitude only.
    const sign = model === "standard" ? (type === "C" ? 1 : -1) : -1;
    const gex = sign * gamma * oi * mult;
    // Vanna (dDelta/dVol) and dealer delta from Black-Scholes with r=0, using the quoted IV.
    const T = Math.max(0.5, dte(exp)) / 365, sig = iv > 0 ? iv : 0.2, sq = sig * Math.sqrt(T);
    const d1 = (Math.log(spot / K) + 0.5 * sig * sig * T) / sq, d2 = d1 - sq, phi = Math.exp(-d1 * d1 / 2) / Math.sqrt(2 * Math.PI);
    const vanna = -phi * d2 / sig;                        // per 1.0 vol (100 pts); same for calls and puts
    const vex = sign * vanna * oi * 100 * spot * 0.01;    // $ delta change per 1-pt move in IV, dealer sign
    const dex = -delta * oi * 100 * spot;                 // dealer delta $ (dealers hold the other side of customer longs)
    contracts.push({exp, type, K, oi, vol, iv, gamma, delta, mid:(bid+ask)/2 || last, gex, vex, dex, vanna});
    if (K < lo || K > hi) continue;
    const key = exp + "|" + K;
    const c = byCell.get(key) || {exp, K, gex:0, vex:0, dex:0, cOI:0, pOI:0, cVol:0, pVol:0, cG:0, pG:0};
    c.gex += gex; c.vex += vex; c.dex += dex; if (type==="C"){ c.cOI+=oi; c.cVol+=vol; c.cG += gamma*oi*mult; } else { c.pOI+=oi; c.pVol+=vol; c.pG += gamma*oi*mult; }
    byCell.set(key, c);
    const s = byStrike.get(K) || {K, gex:0, vex:0, dex:0, cG:0, pG:0, cOI:0, pOI:0};
    s.gex += gex; s.vex += vex; s.dex += dex; if (type==="C"){ s.cG += gamma*oi*mult; s.cOI += oi; } else { s.pG += gamma*oi*mult; s.pOI += oi; }
    byStrike.set(K, s);
    const e = byExp.get(exp) || {exp, gex:0, vex:0}; e.gex += gex; e.vex += vex; byExp.set(exp, e);
  }
  const strikes = [...byStrike.values()].sort((a,b)=>a.K-b.K);
  const total = strikes.reduce((a,s)=>a+s.gex,0), totalVex = strikes.reduce((a,s)=>a+s.vex,0), totalDex = strikes.reduce((a,s)=>a+s.dex,0);
  const vexPeak = strikes.reduce((a,s)=> Math.abs(s.vex) > Math.abs(a?.vex||0) ? s : a, null);
  const callWall = strikes.reduce((a,s)=> s.K > spot*1.002 && s.cG > (a?.cG||0) ? s : a, null);
  const putWall = strikes.reduce((a,s)=> s.K < spot*0.998 && s.pG > (a?.pG||0) ? s : a, null);
  // Gamma flip: strike where cumulative net GEX (from the low end) changes sign, nearest to spot.
  let cum = 0, flip = null, best = Infinity, prev = null;
  for (const s of strikes){ cum += s.gex; if (prev !== null && Math.sign(cum) !== Math.sign(prev) && prev !== 0){ if (Math.abs(s.K - spot) < best){ best = Math.abs(s.K-spot); flip = s.K; } } prev = cum; }
  // Max pain for nearest expiry
  const near = expsSel[0];
  const nearC = contracts.filter(c => c.exp === near);
  const ks = [...new Set(nearC.map(c=>c.K))].sort((a,b)=>a-b);
  let maxPain = null, minPain = Infinity;
  for (const K of ks){ let pain = 0; for (const c of nearC){ pain += c.oi * 100 * (c.type==="C" ? Math.max(0, K - c.K) : Math.max(0, c.K - K)); } if (pain < minPain){ minPain = pain; maxPain = K; } }
  // Expected move per expiry from ATM straddle
  const em = expsSel.map(exp => {
    const cs = contracts.filter(c => c.exp === exp); if (!cs.length) return null;
    const atm = cs.reduce((a,c)=> Math.abs(c.K-spot) < Math.abs(a.K-spot) ? c : a, cs[0]).K;
    const call = cs.find(c=>c.K===atm&&c.type==="C"), put = cs.find(c=>c.K===atm&&c.type==="P");
    const straddle = (call?.mid||0) + (put?.mid||0);
    return {exp, atm, move:straddle, pct:straddle/spot*100, iv:call?.iv||0, dte:dte(exp)};
  }).filter(Boolean);
  // Top nodes
  const cells = [...byCell.values()].sort((a,b)=>Math.abs(b.gex)-Math.abs(a.gex));
  const top = cells.slice(0, 5);
  // Fresh positioning: volume vs OI
  const fresh = contracts.filter(c => c.vol >= S.set.freshMinVol && c.oi > 0 && c.vol/c.oi >= S.set.freshRatio).map(c => ({...c, ratio:c.vol/c.oi, notional:c.vol*c.mid*100})).sort((a,b)=>b.notional-a.notional).slice(0, 40);
  return { spot, strikes, byCell, byExp, allContracts, total, totalVex, totalDex, vexPeak, callWall, putWall, flip, maxPain, near, em, top, fresh, contracts, maxAbs: Math.max(1, ...cells.map(c=>Math.abs(c.gex))) };
}

// Charm: how much delta dealers must trade as one day of time passes on the near expiries (0–3 DTE). Vanna: per 1-pt drop in IV.
function hedgeFlows(){
  const c = S.calc, spot = c.spot; let charm = 0;
  for (const x of c.allContracts){ const d = dte(x.exp); if (d > 3 || !x.iv) continue; const T0 = Math.max(0.25, d)/365, T1 = Math.max(0.05, d-1)/365;
    const delta = (T) => { const sq = x.iv*Math.sqrt(T); const d1 = (Math.log(spot/x.K) + 0.5*x.iv*x.iv*T)/sq; const Nd = 0.5*(1+erf(d1/Math.SQRT2)); return x.type === "C" ? Nd : Nd - 1; };
    const sign = S.set.model === "standard" ? (x.type === "C" ? -1 : 1) : 1; // dealers hold the opposite of the customer
    charm += sign * (delta(T1) - delta(T0)) * x.oi * 100 * spot; }
  return { charm, vanna: c.totalVex };
}

function readText(){

  const regimePos = c.total > 0;
  out.push(regimePos
    ? `${sym} is in positive gamma across the selected expirations. Dealers are net long gamma, so their hedging leans against moves: rallies get sold into and dips get bought. Expect ranges to hold and mean reversion to work better than breakouts.`
    : `${sym} is in negative gamma. Dealers are net short gamma, so their hedging amplifies moves: they buy strength and sell weakness. Expect wider ranges, faster moves and breakouts that follow through.`);
  if (c.flip) out.push(sp > c.flip ? `Spot is ${fmt((sp/c.flip-1)*100,2)}% above the gamma flip at ${fmt(c.flip)}. Reclaiming it would calm things down; until then volatility stays elevated.`);
  if (c.callWall) out.push(`The call wall sits at ${fmt(c.callWall.K)} (${money(c.callWall.cG)} of call gamma). That's the level most likely to cap a rally today; a clean break above it can squeeze as dealers chase.`);
  if (c.putWall) out.push(`The put wall is ${fmt(c.putWall.K)} (${money(c.putWall.pG)} of put gamma), the first real support. Losing it tends to accelerate selling.`);
  out.push(c.totalVex > 0 ? `Vanna is positive: if implied vol comes in, dealers buy back delta and price gets a tailwind${c.vexPeak ? `, strongest around ${fmt(c.vexPeak.K)}` : ""}. That's the mechanism behind post-event grinds higher — moves that get sold until vol actually resets.` : `Vanna is negative: a drop in implied vol makes dealers sell delta, so calmer tape doesn't automatically mean higher prices here.`);
  if (c.em[0]) out.push(`The market is pricing about ±${fmt(c.em[0].move,1)} (${fmt(c.em[0].pct,2)}%) into ${c.em[0].exp}${c.em[0].dte===0?" (0DTE)":""}. Max pain for that expiry is ${fmt(c.maxPain)}.`);

}

  FOMC_2026.forEach(d => out.push({name:"FOMC decision", date:d, auto:true}));
  CPI_2026_APPROX.forEach(d => out.push({name:"CPI (approx.)", date:d, auto:true})); for (let m = 0; m < 3; m++){ const y = now.getFullYear(), mo = now.getMonth()+m; const d1 = new Date(y, mo, 1); const firstFri = new Date(y, mo, 1 + ((5 - d1.getDay() + 7) % 7)); const thirdFri = new Date(firstFri); thirdFri.setDate(firstFri.getDate()+14); out.push({name:"Jobs report (NFP)", date:firstFri.toISOString().slice(0,10), auto:true}, {name:"Monthly opex", date:thirdFri.toISOString().slice(0,10), auto:true}); } return out; }
function nextEventBefore(exp){ const today = new Date(); today.setHours(0,0,0,0); const all = [...autoEvents(), ...getEvents()].map(e => ({...e, days: Math.round((new Date(e.date+"T00:00:00") - today)/86400000)})).filter(e => e.days >= 0 && (!exp || e.date <= exp)).sort((a,b)=>a.days-b.days); return all[0] || null; }
function renderEvents(){
  const el = document.getElementById("events"); if (!el) return; el.replaceChildren();
  const today = new Date(); today.setHours(0,0,0,0);
  const all = [...autoEvents(), ...getEvents()].map(e => ({...e, days: Math.round((new Date(e.date+"T00:00:00") - today)/86400000)})).filter(e => e.days >= -1).sort((a,b)=>a.days-b.days).slice(0,7);
  const isFri = today.getDay() === 5;
  if (isFri) el.append(h("div",{class:"notice", style:{marginBottom:"8px", color:"var(--gold)"}},"Friday: short-dated premium pays for the weekend. Theta × 3 on anything you hold past the close."));
  el.append(h("div",{style:{display:"grid",gap:"2px"}}, ...all.map(e => h("div",{style:{display:"flex",justifyContent:"space-between",gap:"8px",padding:"4px 0",borderBottom:"1px solid var(--rule)",fontSize:"12.5px"}}, h("span",null, e.name, e.auto ? h("small",{class:"hint"}," auto") : null), h("span",{class:"num", style:{color: e.days <= 1 ? "var(--neg)" : e.days <= 5 ? "var(--gold)" : "var(--muted)"}}, e.days === 0 ? "today" : e.days === 1 ? "tomorrow" : e.days < 0 ? "yesterday" : `in ${e.days}d`)))));
  el.append(h("div",{class:"actions",style:{marginTop:"8px"}}, h("button",{class:"btn small", onclick:()=>{ const name = prompt("Event (e.g. CPI, FOMC, NVDA earnings):"); if (!name) return; const date = prompt("Date (YYYY-MM-DD):"); if (!/^\d{4}-\d{2}-\d{2}$/.test(date||"")) return; setEvents([...getEvents(), {name, date}]); renderEvents(); }},"+ Add event"), getEvents().length ? h("button",{class:"btn small", onclick:()=>{ const l = getEvents(); const pick = prompt("Remove which? " + l.map((e,i)=>`${i+1}) ${e.name} ${e.date}`).join("  ")); const i = Number(pick)-1; if (l[i]){ l.splice(i,1); setEvents(l); renderEvents(); } }},"Remove") : null));
}
