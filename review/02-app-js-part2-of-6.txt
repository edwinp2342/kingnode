function checkFlowAlerts(){
  const r = getRules(); if (!r.on || !S.flow || S.flowDemo) return;
  const c = S.calc; const hits = [];
  for (const t of S.flow.trades){ const id = `${t.t}|${t.o||t.K}|${t.sz}`; if (S.flowAlerted.has(id)) continue; S.flowAlerted.add(id);
    if (Date.now() - t.t > 15*60e3) continue; if (t.prem < r.minPrem) continue; if (r.sweepsOnly && t.kind !== "sweep") continue; if (r.sideAsk && t.side !== "A") continue; if (dte(t.exp) < r.minDte) continue;
    const ct = c?.contracts.find(x => x.exp===t.exp && x.K===t.K && x.type===t.cp); const volOi = ct && ct.oi ? ct.vol/ct.oi : 0; if (r.minVolOi && volOi && volOi < r.minVolOi) continue;
    hits.push({t, volOi}); }
  for (const {t, volOi} of hits.slice(0, 5)){ const text = `${t.u} ${fmt(t.K)}${t.cp} ${t.exp} — ${money(t.prem)} ${t.kind.toUpperCase()} at the ${t.side==="A"?"ASK":t.side==="B"?"BID":"MID"}, ${fmt(t.sz)} contracts${volOi ? `, vol/OI ${fmt(volOi,1)}×` : ""}`; fireAlert("Suspicious flow", text); }
}

function heatColor(v, max){
  const t = Math.min(1, Math.abs(v)/max); const a = 0.08 + 0.85*Math.sqrt(t);
  return v >= 0 ? `rgba(45,212,191,${a})` : `rgba(249,112,102,${a})`;
}
const S_HEAT = { mode:"smooth" };
function heatRamp(t){ // diverging: coral (−) → dark → teal (+), t in [-1,1]
  const a = Math.min(1, Math.abs(t)); const ease = Math.pow(a, 0.6);
  if (t >= 0) return `rgba(38,166,154,${0.06 + 0.9*ease})`;
  return `rgba(239,83,80,${0.06 + 0.9*ease})`;
}
function viewHeatSmooth(){
  const c = S.calc, exps = S.exps, strikes = c.strikes.map(s=>s.K).sort((a,b)=>b-a);
  if (!strikes.length) return h("div",{class:"empty"},"No strikes in range.");
  const val = cell => !cell ? 0 : S.hview==="net" ? cell.gex : S.hview==="calls" ? cell.cG : S.hview==="puts" ? -cell.pG : S.hview==="vex" ? cell.vex : S.hview==="dex" ? cell.dex : S.hview==="oi" ? cell.cOI+cell.pOI : cell.cVol+cell.pVol;
  const mag = S.hview==="oi"||S.hview==="vol";
  const grid = exps.map(e => strikes.map(K => val(c.byCell.get(e+"|"+K))));
  // vertical smoothing across strikes (gaussian, 5 taps) so nodes read as bands, not blocks
  const kern = [0.06,0.24,0.4,0.24,0.06];
  const sm = grid.map(col => col.map((_,i) => kern.reduce((a,w,k) => a + w*(col[i+k-2] ?? col[i]), 0)));
  const vmax = Math.max(1e-9, ...sm.flat().map(Math.abs));
  const availW = Math.max(600, (document.getElementById("panel")?.clientWidth || 900) - 24);
  const L = 78, T = 34, cw = Math.max(70, Math.floor((availW - L - 8)/Math.max(1,exps.length))), rh = 20;
  const W = L + exps.length*cw + 8, H = T + strikes.length*rh + 10, dpr = Math.min(2, window.devicePixelRatio||1);
  const cv = h("canvas",{width:W*dpr, height:H*dpr, style:{width:W+"px", height:H+"px", display:"block", cursor:"pointer"}});
  const g = cv.getContext("2d"); g.scale(dpr,dpr);
  const css = n => getComputedStyle(document.documentElement).getPropertyValue(n).trim();
  g.fillStyle = css("--bg"); g.fillRect(0,0,W,H);
  // cells: vertical gradient between neighbours for a continuous look
  exps.forEach((e,ci) => { const x = L + ci*cw;
    strikes.forEach((K,ri) => { const y = T + ri*rh; const v = sm[ci][ri]/vmax, vn = (sm[ci][ri+1] ?? sm[ci][ri])/vmax;
      const gr = g.createLinearGradient(0,y,0,y+rh); gr.addColorStop(0, heatRamp(mag ? Math.abs(v) : v)); gr.addColorStop(1, heatRamp(mag ? Math.abs(vn) : vn));
      g.fillStyle = gr; g.fillRect(x+1, y, cw-2, rh); }); });
  // labels
  g.font = "11px " + css("--mono"); g.textBaseline = "middle";
  const spotIdx = strikes.findIndex(k => k <= c.spot);
  const em = c.em[0], emLo = em ? c.spot-em.move : null, emHi = em ? c.spot+em.move : null;
  strikes.forEach((K,ri) => { const y = T + ri*rh + rh/2; const inEm = emLo != null && K >= emLo && K <= emHi;
    g.fillStyle = ri===spotIdx ? css("--spot") : inEm ? css("--ink") : css("--muted"); g.textAlign = "right"; g.fillText(fmt(K), L-8, y);
    if (inEm){ g.fillStyle = css("--gold"); g.fillRect(L-4, T + ri*rh, 2, rh); } });
  g.textAlign = "center"; g.fillStyle = css("--muted");
  exps.forEach((e,ci) => { const x = L + ci*cw + cw/2; g.fillText(e.slice(5) + (dte(e)===0?" 0D":""), x, 12); g.fillText(dte(e)+"d", x, 25); });
